find_package(Qt6 REQUIRED COMPONENTS Test)
set(CMAKE_AUTOMOC ON)
function(prueba nombre)
    add_executable(${nombre} ${nombre}.cpp)
    target_link_libraries(${nombre} PRIVATE Qt6::Test ${ARGN})
    add_test(NAME ${nombre} COMMAND ${nombre})
    set_tests_properties(${nombre} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endfunction()
prueba(tst_cripto comun_sinc comun_monocypher)
target_include_directories(tst_cripto PRIVATE ../terceros/monocypher)
prueba(tst_sinc comun_sinc)
